CREATE TABLE public.oportunidades (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  nombre text NOT NULL CHECK (char_length(nombre) >= 2 AND char_length(nombre) <= 120),
  empresa text,
  email text NOT NULL CHECK (email ~* '^[^@\s]+@[^@\s]+\.[^@\s]+$'::text),
  telefono text,
  servicio text NOT NULL,
  mensaje text CHECK (char_length(mensaje) <= 2000),
  estado text NOT NULL DEFAULT 'nueva'::text CHECK (estado = ANY (ARRAY['nueva'::text, 'contactada'::text, 'propuesta'::text, 'ganada'::text, 'perdida'::text])),
  valor_estimado numeric,
  empleado_id uuid UNIQUE,
  CONSTRAINT oportunidades_pkey PRIMARY KEY (id),
  CONSTRAINT oportunidades_empleado_id_fkey FOREIGN KEY (empleado_id) REFERENCES public.empleados(id)
);
CREATE TABLE public.empresas (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  nombre text NOT NULL CHECK (char_length(nombre) >= 2 AND char_length(nombre) <= 120),
  ubicacion text NOT NULL CHECK (char_length(ubicacion) >= 2 AND char_length(ubicacion) <= 160),
  CONSTRAINT empresas_pkey PRIMARY KEY (id)
);
CREATE TABLE public.empleados (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  created_at timestamp with time zone NOT NULL DEFAULT now(),

  empresa_id uuid,
